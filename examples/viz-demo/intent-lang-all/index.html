147@doc("载入待验证的需求文件，进入 Unverified")

154@doc("解析 .intent 源文件为 AST")
155intent ParseModelSpec(m: ModelSpec) {
156 modifies m.phase
157
158 require from: m.phase == Unverified else reject
159 ensure parsed: m.phase' == Parsed

162@asis
163@doc("类型检查通过后才进入结构分析")
164intent TypeCheckModelSpec(m: ModelSpec) {
165 modifies m.phase
166
167 require from: m.phase == Parsed else reject
168 ensure typed: m.phase' == TypeChecked
169}

171@asis
172@doc("结构门控：goal 认领、example、生命周期可达性（S0001–S0008）")
173intent CheckStructureModelSpec(m: ModelSpec) {
174 modifies m.phase, m.structurallySound
175
176 require from: m.phase == TypeChecked else reject
177 ensure checked: m.phase' == StructureChecked
178 ensure sound: m.structurallySound' == true
179}

181@asis
182@doc("Z3 验证 intent/theorem/example；@asis 规则默认跳过除非 --include-asis")
183intent VerifyModelSpec(m: ModelSpec) {
184 modifies m.phase, m.logicallyConsistent
185
186 require from: m.phase == StructureChecked else reject
187 ensure verified: m.phase' == Verified
188 ensure consistent: m.logicallyConsistent' == true
189}

191@asis
192@doc("V0020：规则子句不可满足时从 StructureChecked 转入 Failed")
193intent RejectVacuousRules(m: ModelSpec) {
194 modifies m.phase, m.logicallyConsistent
195
196 require from: m.phase == StructureChecked else reject
197 require vacuous: !m.logicallyConsistent else reject
198 ensure failed: m.phase' == Failed
199}

255@tobe
256@doc("accept 管线在 Z3/结构门通过后才允许 gen（当前实现缺口，见 SUS）")
257intent GateAcceptanceOnVerification(m: ModelSpec, b: FactsBaseline) {
258 require verified: m.phase == Verified else reject
259 require formalized: b.phase == Formalized else reject
260 ensure gated: b.phase' == b.phase
261}

263@asis
264@doc("accept run：strict gate 下 failed 或 manual-pending 则 fail")
265intent RunAcceptanceGate(j: AcceptanceJob) {
266 modifies j.allMachinePassed
267
268 require strict: j.strictGate else reject
269 require no_manual: j.manualPending == 0 else reject
270 ensure passed: j.allMachinePassed' == true
271}
272
273@asis
274@doc("lenient gate：仅有 manual-pending 无 failed 时可 pass-with-pending")
275intent RunAcceptanceLenient(j: AcceptanceJob) {
276 modifies j.allMachinePassed
277
278 require lenient: !j.strictGate else reject
279 require pending_ok: j.manualPending >= 0 else reject
280 ensure passed: j.allMachinePassed' == true
281}
