160@asis
161@doc("解析连接串、选择驱动并 connect，成功则进入 Connected")
162intent EstablishConnection(s: DeviceSession, a: ConnectionAttempt) {
163 modifies s.phase
164
165 require uri_syntax: a.uriSyntaxValid else reject // F-P4X-BEH-001
166 require protocol: a.protocolRegistered else reject // F-P4X-BEH-002
167 require pattern: a.uriPatternMatch else reject // F-P4X-BEH-003
168 require transport: a.hasDefaultOrExplicitTransport else reject // F-P4X-BEH-004
169 require driver_match: a.driverProtocolMatch else reject // F-P4X-BEH-005
170 require transport_ok: a.transportSupported else reject // F-P4X-BEH-006
171 require transport_reg: a.transportRegistered else reject // F-P4X-BEH-007
172 require disconnected: s.phase == Disconnected else reject
173
174 ensure connected: s.phase' == Connected // F-P4X-BEH-008 F-P4X-BEH-038
175 // F-P4X-BEH-009: 同一 protocol code 多驱动 → IllegalStateException（ServiceLoader 注册时）
176}
177
178@asis
179@doc("网络异常导致连接意外断开")

181 modifies s.phase
182
183 require connected: s.phase == Connected else reject
184 ensure lost: s.phase' == ConnectionLost
185}

189intent CloseConnection(s: DeviceSession) {

193 ensure disconnected: s.phase' == Disconnected
194}

205@asis
206@doc("任一连接校验失败时拒绝，phase 保持 Disconnected")
207intent RejectBadConnection(s: DeviceSession, a: ConnectionAttempt) {
208 modifies s.phase
209
210 require any_bad:
211 !a.uriSyntaxValid ||
212 !a.protocolRegistered ||
213 !a.uriPatternMatch ||
214 !a.hasDefaultOrExplicitTransport ||
215 !a.driverProtocolMatch ||
216 !a.transportSupported ||
217 !a.transportRegistered else reject
218 require disconnected: s.phase == Disconnected else reject
219
220 ensure stay_off: s.phase' == Disconnected // F-P4X-BEH-010
221}

225@asis
226@doc("已连接会话批量读取 tag；重复名在 build 阶段拒绝")
227intent ReadTags(s: DeviceSession, r: TagRequest) {
228 modifies s.phase
229
230 require connected: s.phase == Connected else reject
231 require no_dup: !r.duplicateName else reject // F-P4X-BEH-011
232 require handler: r.tagHandlerPresent else reject // F-P4X-BEH-012
233
234 ensure stay_connected: s.phase' == Connected
235 // F-P4X-BEH-013: 无效地址 → per-tag INVALID_ADDRESS（build 软失败，非 reject）
236 // F-P4X-BEH-014: 驱动不支持 → execute failedFuture（运行时，非状态迁移）
237 // F-P4X-BEH-015: 同一 PlcReadRequest 可含多个 tag name，批量读取
238 // F-P4X-BEH-016: getPlcValue 对非 OK tag 抛出 PlcRuntimeException
239}

241@asis
242@doc("已连接会话批量写入 tag；重复名或空值在 build 阶段拒绝")
243intent WriteTags(s: DeviceSession, r: TagRequest) {
244 modifies s.phase
245
246 require connected: s.phase == Connected else reject
247 require no_dup: !r.duplicateName else reject // F-P4X-BEH-017
248 require has_values: r.hasValues else reject // F-P4X-BEH-018
249
250 ensure stay_connected: s.phase' == Connected
251 // F-P4X-BEH-019: 无效地址/数据 → per-tag INVALID_ADDRESS/INVALID_DATA
252 // F-P4X-BEH-020: build-time 错误 tag 不写入设备
253 // F-P4X-BEH-021: 驱动未实现 onWrite → execute failedFuture
254}

285@doc("从 Open 池中按连接串租约连接")
286intent AcquireCachedConnection(p: ConnectionPool) {
287 modifies p.phase, p.leaseAcquired
288
289 require pool_open: p.phase == Open else reject // F-P4X-BEH-028

300@asis
301@doc("触发器触发时批量读取 tag；guard 不通过则跳过")
302intent FetchTagBatch(b: TagBatch) {
303 modifies b.phase, b.fetchInProgress
304
305 require active: b.phase == Active else reject // F-P4X-BEH-032 对立路径
306 require not_busy: !b.fetchInProgress else reject // F-P4X-BEH-035 简化：in-progress 拒绝新 fetch
307 require not_backoff: !b.inBackoff else reject // F-P4X-BEH-034
308
309 ensure stay_active: b.phase' == Active
310 ensure fetching: b.fetchInProgress' == true
311 // F-P4X-BEH-032: Closed batch → require active else reject
312 // F-P4X-BEH-033: 无 tags 时 fetchTags 跳过（completedFuture null）
313 // F-P4X-BEH-036: tag 集合不变时缓存 PlcReadRequest 跨轮询复用
314}
